# Regression cases: run in parallel, each in its own empty dir
# $WORK/cases/<name>/ (cwd), with ./data -> input data and ./fx -> fixtures.
# stdout/stderr/exit code of the command plus every regular file the case
# leaves in its dir are compared byte-for-byte against golden/cases/<name>/
# (after the volatile-line normalization described in dev/README.md).
#
# Format: <name> <shell command...>   (first token = name, rest = command)
# Keep outputs small: delete big intermediates (e.g. FM-index dbs) in the command.

# --dbcache (targets read once, kept in memory): same results as without it;

hmmsearch_dbcache_gz_multi gzip -nc data/protdb.fa > db.fa.gz && hmmsearch --dbcache --cpu 2 --tblout t.tbl -o main.out fx/minipfam.hmm db.fa.gz; rc=$?; rm -f db.fa.gz; exit $rc
hmmsearch_dbcache_stdin_multi cat data/protdb.fa | hmmsearch --dbcache --cpu 0 --tblout t.tbl -o main.out fx/minipfam.hmm -
